  105  // Separating axis test for triangle-box overlap
  106  bool triangleBoxOverlap(const Vec3& boxCenter, const Vec3& boxHalfSize, const Triangle& tri) {
  107      // Translate triangle to box center
  108      Vec3 v0 = tri.a - boxCenter;
  109      Vec3 v1 = tri.b - boxCenter;
  110      Vec3 v2 = tri.c - boxCenter;
  111  
  112      // Triangle edges
  113      Vec3 e0 = v1 - v0;
  114      Vec3 e1 = v2 - v1;
  115      Vec3 e2 = v0 - v2;
  116  
  117      // Test box normals (AABB axes)
  118      auto minMax3 = [](val_t a, val_t b, val_t c) {
  119          return std::make_pair(std::min({a, b, c}), std::max({a, b, c}));
  120      };
  121  
  122      auto [minX, maxX] = minMax3(v0.x, v1.x, v2.x);
  123      if (minX > boxHalfSize.x || maxX < -boxHalfSize.x) return false;
  124  
  125      auto [minY, maxY] = minMax3(v0.y, v1.y, v2.y);
  126      if (minY > boxHalfSize.y || maxY < -boxHalfSize.y) return false;
  127  
  128      auto [minZ, maxZ] = minMax3(v0.z, v1.z, v2.z);
  129      if (minZ > boxHalfSize.z || maxZ < -boxHalfSize.z) return false;
  130  
  131      // Test triangle normal
  132      Vec3 triNormal = e0.cross(e1);
  133      val_t d = triNormal.dot(v0);
  134      val_t r = boxHalfSize.x * std::abs(triNormal.x) +
  135                boxHalfSize.y * std::abs(triNormal.y) +
  136                boxHalfSize.z * std::abs(triNormal.z);
  137      if (std::abs(d) > r) return false;

  141          val_t p0 = axis.dot(v0);
  142          val_t p1 = axis.dot(v1);
  143          val_t p2 = axis.dot(v2);
  144          val_t r = boxHalfSize.x * std::abs(axis.x) +
  145                    boxHalfSize.y * std::abs(axis.y) +
  146                    boxHalfSize.z * std::abs(axis.z);

  148          return !(minP > r || maxP < -r);
  149      };

  285          if (std::abs(d.y * c.z - d.z * c.y) > halfExtent.y * ad.z + halfExtent.z * ad.y + EPSILON) return false;
  286          if (std::abs(d.z * c.x - d.x * c.z) > halfExtent.z * ad.x + halfExtent.x * ad.z + EPSILON) return false;
  287          if (std::abs(d.x * c.y - d.y * c.x) > halfExtent.x * ad.y + halfExtent.y * ad.x + EPSILON) return false;
  288  
  289          return true;
  290      }
  291  
  292      // Apply a function to all triangles potentially intersecting the ray
  293      // Returns true if the function returns true for any triangle
  294      template<typename Func>
  295      bool applyToTris(const Vec3& p1, const Vec3& p2, Func&& func) const {
  296          // If leaf node, check triangles directly
  297          if (!triangleIndices.empty()) {
  298              for (size_t idx : triangleIndices) {
  299                  if (func(idx, (*allTriangles)[idx])) return true;
  300              }
  301              return false;
  302          }
  303  
  304          // Otherwise, descend to children
  305          for (int i = 0; i < 8; ++i) {
  306              if (children[i] && children[i]->rayIntersectsBox(p1, p2)) {
  307                  if (children[i]->applyToTris(p1, p2, func)) return true;
  308              }
  309          }
  310          return false;
  311      }
  312  };

  315  // Mesh Generation: Icosphere
  316  // ============================================================================

  318  // Generate an icosphere by subdividing an icosahedron
  319  class IcosphereMesh {
  320  public:
  321      std::vector<Triangle> triangles;
  322  
  323      IcosphereMesh(int subdivisions, val_t radius) {
  324          // Initial icosahedron vertices
  325          const val_t t = (1.0f + std::sqrt(5.0f)) / 2.0f;
  326  
  327          std::vector<Vec3> vertices = {
  328              Vec3(-1,  t,  0).normalized() * radius,
  329              Vec3( 1,  t,  0).normalized() * radius,
  330              Vec3(-1, -t,  0).normalized() * radius,
  331              Vec3( 1, -t,  0).normalized() * radius,
  332              Vec3( 0, -1,  t).normalized() * radius,
  333              Vec3( 0,  1,  t).normalized() * radius,
  334              Vec3( 0, -1, -t).normalized() * radius,
  335              Vec3( 0,  1, -t).normalized() * radius,
  336              Vec3( t,  0, -1).normalized() * radius,
  337              Vec3( t,  0,  1).normalized() * radius,
  338              Vec3(-t,  0, -1).normalized() * radius,
  339              Vec3(-t,  0,  1).normalized() * radius
  340          };
  341  
  342          // Initial icosahedron faces (20 triangles)
  343          std::vector<std::array<idx_t, 3>> faces = {
  344              {0, 11, 5}, {0, 5, 1}, {0, 1, 7}, {0, 7, 10}, {0, 10, 11},
  345              {1, 5, 9}, {5, 11, 4}, {11, 10, 2}, {10, 7, 6}, {7, 1, 8},
  346              {3, 9, 4}, {3, 4, 2}, {3, 2, 6}, {3, 6, 8}, {3, 8, 9},
  347              {4, 9, 5}, {2, 4, 11}, {6, 2, 10}, {8, 6, 7}, {9, 8, 1}
  348          };
  349  
  350          // Subdivide
  351          for (int i = 0; i < subdivisions; ++i) {
  352              std::vector<std::array<idx_t, 3>> newFaces;
  353              std::map<std::pair<idx_t, idx_t>, idx_t> midpointCache;
  354  
  355              auto getMidpoint = [&](idx_t i1, idx_t i2) -> idx_t {
  356                  auto key = std::make_pair(std::min(i1, i2), std::max(i1, i2));
  357                  auto it = midpointCache.find(key);
  358                  if (it != midpointCache.end()) return it->second;
  359  
  360                  Vec3 mid = (vertices[i1] + vertices[i2]) / 2.0f;
  361                  mid = mid.normalized() * radius;
  362                  idx_t idx = static_cast<idx_t>(vertices.size());
  363                  vertices.push_back(mid);
  364                  midpointCache[key] = idx;
  365                  return idx;
  366              };
  367  
  368              for (const auto& face : faces) {
  369                  idx_t a = getMidpoint(face[0], face[1]);
  370                  idx_t b = getMidpoint(face[1], face[2]);
  371                  idx_t c = getMidpoint(face[2], face[0]);
  372  
  373                  newFaces.push_back({face[0], a, c});
  374                  newFaces.push_back({face[1], b, a});
  375                  newFaces.push_back({face[2], c, b});
  376                  newFaces.push_back({a, b, c});
  377              }
  378              faces = std::move(newFaces);
  379          }
  380  
  381          // Build triangles (normals pointing inward for a "room")
  382          triangles.reserve(faces.size());
  383          for (const auto& face : faces) {
  384              // Reverse winding to make normals point inward
  385              triangles.emplace_back(vertices[face[2]], vertices[face[1]], vertices[face[0]]);
  386          }
  387      }
  388  };

  390  // ============================================================================
  391  // Random Number Generation
  392  // ============================================================================
  393  
  394  class RandomGenerator {
  395      std::mt19937 rng;
  396      std::uniform_real_distribution<val_t> dist;
  397  public:
  398      explicit RandomGenerator(uint32_t seed = 42) : rng(seed), dist(0.0f, 1.0f) {}
  399      val_t rand() { return dist(rng); }
  400  };
  401  
  402  // Generate a random point inside a triangle using barycentric coordinates
  403  Vec3 randomPointInTriangle(const Triangle& t, RandomGenerator& rng) {
  404      val_t u = rng.rand();
  405      val_t v = rng.rand();
  406      if (u + v > 1.0f) {
  407          u = 1.0f - u;
  408          v = 1.0f - v;
  409      }
  410      Vec3 ab = t.b - t.a;
  411      Vec3 ac = t.c - t.a;
  412      return t.a + ab * u + ac * v;
  413  }

  415  // ============================================================================
  416  // Ray-Triangle Intersection (Möller-Trumbore algorithm)
  417  // ============================================================================
  418  
  419  val_t rayTriangleIntersect(const Vec3& orig, const Vec3& dir,
  420                              const Vec3& v0, const Vec3& v1, const Vec3& v2) {
  421      Vec3 e1 = v1 - v0;
  422      Vec3 e2 = v2 - v0;
  423      Vec3 pvec = dir.cross(e2);
  424      val_t det = e1.dot(pvec);
  425  
  426      if (std::abs(det) < EPSILON) return std::numeric_limits<val_t>::max();
  427  
  428      val_t invDet = 1.0f / det;
  429      Vec3 tvec = orig - v0;
  430      val_t u = tvec.dot(pvec) * invDet;
  431      if (u < 0.0f || u > 1.0f) return std::numeric_limits<val_t>::max();
  432  
  433      Vec3 qvec = tvec.cross(e1);
  434      val_t v = dir.dot(qvec) * invDet;
  435      if (v < 0.0f || u + v > 1.0f) return std::numeric_limits<val_t>::max();
  436  
  437      return e2.dot(qvec) * invDet;
  438  }

  440  // ============================================================================
  441  // Visibility Testing (Octree-accelerated)
  442  // ============================================================================
  443  
  444  // Check if a ray between two triangles is blocked by any other triangle
  445  // Uses octree for O(log N) instead of O(N) search
  446  bool isRayBlocked(const Vec3& from, const Vec3& to,
  447                    const Octree& octree,
  448                    size_t srcTriIdx, size_t dstTriIdx) {
  449      Vec3 dir = to - from;
  450      val_t rayLen = dir.norm();
  451      if (rayLen < EPSILON) return true;
  452      Vec3 dirNorm = dir / rayLen;
  453  
  454      return octree.applyToTris(from, to, [&](size_t idx, const Triangle& tri) {
  455          if (idx == srcTriIdx || idx == dstTriIdx) return false;
  456  
  457          val_t dist = rayTriangleIntersect(from, dirNorm, tri.a, tri.b, tri.c);
  458          if (dist > EPSILON && dist < rayLen - EPSILON) {
  459              return true;  // Ray is blocked
  460          }
  461          return false;
  462      });
  463  }

  465  // ============================================================================
  466  // Form Factor Computation (Kij)
  467  // ============================================================================
  468  
  469  // Compute the cosine of angle between vector and triangle normal
  470  val_t cosPhi(const Vec3& v, const Vec3& normal) {
  471      val_t vNorm = v.norm();
  472      if (vNorm <= EPSILON) return ZERO;
  473      return std::max(ZERO, v.dot(normal) / vNorm);
  474  }
  475  
  476  // Compute form factor Kij between triangle i (receiver) and triangle j (emitter)
  477  val_t computeKij(size_t idxI, size_t idxJ,
  478                    const std::vector<Triangle>& triangles,
  479                    const Octree& octree,
  480                    RandomGenerator& rng) {
  481      const Triangle& triI = triangles[idxI];
  482      const Triangle& triJ = triangles[idxJ];
  483  
  484      // Cull triangles facing the same direction
  485      if (triI.normal().dot(triJ.normal()) > 0.99f) return ZERO;
  486  
  487      val_t kij = ZERO;
  488  
  489      for (int r = 0; r < NUM_RAYS; ++r) {
  490          Vec3 pI = randomPointInTriangle(triI, rng);
  491          Vec3 pJ = randomPointInTriangle(triJ, rng);
  492  
  493          if (isRayBlocked(pI, pJ, octree, idxI, idxJ)) continue;
  494  
  495          Vec3 v = pJ - pI;
  496          val_t distSqr = v.squaredNorm();
  497          if (distSqr < EPSILON) continue;
  498  
  499          val_t cosPhiI = cosPhi(v, triI.normal());
  500          val_t cosPhiJ = cosPhi(-v, triJ.normal());
  501  
  502          if (cosPhiI <= ZERO || cosPhiJ <= ZERO) continue;
  503  
  504          kij += (cosPhiI * cosPhiJ) / (PI * distSqr);
  505      }
  506  
  507      return kij * INV_NUM_RAYS;
  508  }

  510  // ============================================================================
  511  // Tau (time delay) Computation
  512  // ============================================================================
  513  
  514  int computeTau(const Triangle& triI, const Triangle& triJ) {
  515      val_t dist = (triI.center() - triJ.center()).norm();
  516      return static_cast<int>(std::ceil(dist * INV_WAVE_SPEED));
  517  }

  525      size_t numTimesteps;

  530      std::vector<val_t> kij;         // Form factors (N x N matrix, row-major)
  531      std::vector<int> tau;           // Time delays (N x N matrix, row-major)
  532      std::vector<val_t> radE;        // Emission radiosity (T x N matrix)
  533      std::vector<val_t> radB;        // Reflected radiosity (T x N matrix)

  540      size_t idx2d(size_t i, size_t j) const { return i * numTriangles + j; }
  541      size_t idxTN(size_t t, size_t n) const { return t * numTriangles + n; }
  542  };

  548  void initializeSimulation(SimulationState& state, int subdivisions, size_t timesteps,
  549                            size_t sourceIdx, val_t reflectivity) {
  550      // Generate mesh
  551      IcosphereMesh mesh(subdivisions, 10.0f);  // Radius 10 units
  552      state.triangles = std::move(mesh.triangles);
  553      state.numTriangles = state.triangles.size();
  554      state.numTimesteps = timesteps;
  555      state.sourceIndex = sourceIdx %% state.numTriangles;
  556  
  557      printf("Generated icosphere mesh with %%zu triangles\n", state.numTriangles);
  558  
  559      // Build octree for spatial acceleration
  560      printf("Building octree...\n");
  561      state.octree.build(state.triangles);
  562  
  563      // Initialize areas
  564      state.areas.resize(state.numTriangles);
  565      for (size_t i = 0; i < state.numTriangles; ++i) {
  566          state.areas[i] = state.triangles[i].area();
  567      }
  568  
  569      // Initialize reflectivity
  570      state.rho.resize(state.numTriangles, reflectivity);
  571  
  572      // Initialize matrices
  573      state.kij.resize(state.numTriangles * state.numTriangles, ZERO);
  574      state.tau.resize(state.numTriangles * state.numTriangles, 0);
  575      state.radE.resize(timesteps * state.numTriangles, ZERO);
  576      state.radB.resize(timesteps * state.numTriangles, ZERO);
  577      state.distances.resize(state.numTriangles, ZERO);
  578  
  579      // Set source emission (active for first half of timesteps)
  580      size_t timeOn = 0;
  581      size_t timeOff = timesteps / 2;
  582      for (size_t t = timeOn; t < timeOff; ++t) {
  583          state.radE[state.idxTN(t, state.sourceIndex)] = 1.0f;
  584      }
  585  }

  591  void computeFormFactors(SimulationState& state) {
  592      printf("Computing form factors (Kij)...\n");
  593      RandomGenerator rng(42);
  594  
  595      for (size_t i = 0; i < state.numTriangles; ++i) {
  596          for (size_t j = 0; j < state.numTriangles; ++j) {
  597              if (i == j) continue;
  598              state.kij[state.idx2d(i, j)] = computeKij(
  599                  i, j, state.triangles, state.octree, rng);
  600          }
  601          if ((i + 1) %% 100 == 0 || i + 1 == state.numTriangles) {
  602              printf("  Progress: %%zu/%%zu triangles\n", i + 1, state.numTriangles);
  603          }
  604      }
  605  }

  607  void computeTimeDelays(SimulationState& state) {
  608      printf("Computing time delays (Tau)...\n");
  609  
  610      for (size_t i = 0; i < state.numTriangles; ++i) {
  611          for (size_t j = 0; j < state.numTriangles; ++j) {
  612              if (i == j) continue;
  613              state.tau[state.idx2d(i, j)] = computeTau(
  614                  state.triangles[i], state.triangles[j]);
  615          }
  616      }
  617  }

  619  // ============================================================================
  620  // Simulation Phase (Wave Propagation)
  621  // ============================================================================
  622  
  623  void runSimulation(SimulationState& state) {
  624      printf("Running wave propagation simulation...\n");
  625  
  626      for (size_t t = 0; t < state.numTimesteps; ++t) {
  627          for (size_t i = 0; i < state.numTriangles; ++i) {
  628              val_t sumB = ZERO;
  629  
  630              for (size_t j = 0; j < state.numTriangles; ++j) {
  631                  if (i == j) continue;
  632  
  633                  int tauij = state.tau[state.idx2d(i, j)];
  634  
  635                  // Skip if wave hasn't yet propagated from j to i
  636                  if (static_cast<int>(t) < tauij) continue;
  637  
  638                  val_t kij = state.kij[state.idx2d(i, j)];
  639                  if (kij <= ZERO) continue;
  640  
  641                  // Get radiosity from source triangle at time when emission occurred
  642                  size_t srcTime = t - static_cast<size_t>(tauij);
  643                  val_t radJ = state.radB[state.idxTN(srcTime, j)];
  644                  if (radJ <= ZERO) continue;
  645  
  646                  // Accumulate contribution: form factor * area * source radiosity
  647                  sumB += std::min(kij * state.areas[j], ONE) * radJ;
  648              }
  649  
  650              // Update radiosity: reflection + emission
  651              state.radB[state.idxTN(t, i)] = state.rho[i] * sumB + state.radE[state.idxTN(t, i)];
  652          }
  653  
  654          if ((t + 1) %% 10 == 0 || t + 1 == state.numTimesteps) {
  655              printf("  Timestep %%zu/%%zu\n", t + 1, state.numTimesteps);
  656          }
  657      }
  658  }

  660  // ============================================================================
  661  // Distance Computation (Cross-Correlation)
  662  // ============================================================================
  663  
  664  void computeDistances(SimulationState& state) {
  665      printf("Computing distances via cross-correlation...\n");
  666  
  667      for (size_t i = 0; i < state.numTriangles; ++i) {
  668          val_t maxCorr = ZERO;
  669          int bestT = 0;
  670  
  671          // Discrete cross-correlation to find time delay
  672          for (size_t t = 0; t < state.numTimesteps; ++t) {
  673              val_t sum = ZERO;
  674  
  675              for (size_t tt = t; tt < state.numTimesteps; ++tt) {
  676                  val_t pB = state.radB[state.idxTN(tt, i)];
  677                  val_t pS = state.radB[state.idxTN(tt - t, state.sourceIndex)];
  678                  sum += pS * pB;
  679              }
  680  
  681              if (sum > maxCorr) {
  682                  maxCorr = sum;
  683                  bestT = static_cast<int>(t);
  684              }
  685          }
  686  
  687          state.distances[i] = WAVE_SPEED * static_cast<val_t>(bestT);
  688      }
  689  }

  691  // ============================================================================
  692  // Validation
  693  // ============================================================================
  694  
  695  bool validateResults(const SimulationState& state) {
  696      printf("\nValidation:\n");
  697  
  698      // Check that distances are non-negative
  699      bool allNonNegative = true;
  700      val_t minDist = std::numeric_limits<val_t>::max();
  701      val_t maxDist = std::numeric_limits<val_t>::lowest();
  702      val_t sumDist = ZERO;
  703      int nonZeroCount = 0;
  704  
  705      for (size_t i = 0; i < state.numTriangles; ++i) {
  706          val_t d = state.distances[i];
  707          if (d < 0) {
  708              allNonNegative = false;
  709              printf("  ERROR: Negative distance at triangle %%zu: %%f\n", i, d);
  710          }
  711          if (!std::isfinite(d)) {
  712              printf("  ERROR: Non-finite distance at triangle %%zu: %%f\n", i, d);
  713              return false;
  714          }
  715          minDist = std::min(minDist, d);
  716          maxDist = std::max(maxDist, d);
  717          sumDist += d;
  718          if (d > EPSILON) nonZeroCount++;
  719      }
  720  
  721      printf("  Distance range: [%%.4f, %%.4f]\n", minDist, maxDist);
  722      printf("  Average distance: %%.4f\n", sumDist / static_cast<val_t>(state.numTriangles));
  723      printf("  Non-zero distances: %%d/%%zu\n", nonZeroCount, state.numTriangles);
  724  
  725      // Check source distance is zero or very small
  726      val_t srcDist = state.distances[state.sourceIndex];
  727      if (srcDist > WAVE_SPEED * 2) {
  728          printf("  WARNING: Source triangle distance is non-zero: %%.4f\n", srcDist);
  729      }
  730  
  731      // Check radiosity propagation (some triangles should have received energy)
  732      int receivedEnergy = 0;
  733      for (size_t i = 0; i < state.numTriangles; ++i) {
  734          for (size_t t = 0; t < state.numTimesteps; ++t) {
  735              if (state.radB[state.idxTN(t, i)] > EPSILON) {
  736                  receivedEnergy++;
  737                  break;
  738              }
  739          }
  740      }
  741  
  742      printf("  Triangles receiving energy: %%d/%%zu\n", receivedEnergy, state.numTriangles);
  743  
  744      if (receivedEnergy == 0) {
  745          printf("  ERROR: No triangles received energy - simulation failed\n");
  746          return false;
  747      }
  748  
  749      // Check Kij matrix (should have some non-zero entries)
  750      int nonZeroKij = 0;
  751      for (size_t i = 0; i < state.numTriangles * state.numTriangles; ++i) {
  752          if (state.kij[i] > EPSILON) nonZeroKij++;
  753      }
  754      printf("  Non-zero form factors: %%d/%%zu (%%.2f%%%%)\n",
  755             nonZeroKij, state.numTriangles * state.numTriangles,
  756             100.0f * nonZeroKij / static_cast<val_t>(state.numTriangles * state.numTriangles));
  757  
  758      if (nonZeroKij == 0) {
  759          printf("  ERROR: All form factors are zero - visibility computation failed\n");
  760          return false;
  761      }
  762  
  763      if (!allNonNegative) {
  764          return false;
  765      }
  766  
  767      printf("  Validation: PASSED\n");
  768      return true;
  769  }

  775  uint64_t computeHash(const SimulationState& state) {
  776      uint64_t hash = 0;
  777      for (size_t i = 0; i < state.numTriangles; ++i) {
  778          const uint32_t* ptr = reinterpret_cast<const uint32_t*>(&state.distances[i]);
  779          hash ^= (static_cast<uint64_t>(*ptr) + i) * 0x9e3779b97f4a7c15ULL;
  780      }
  781      return hash;
  782  }

  832          } else if (strcmp(argv[i], "-s") == 0 && i + 1 < argc) {
  833              sourceIdx = atoi(argv[++i]);
  834          } else if (strcmp(argv[i], "-r") == 0 && i + 1 < argc) {
  835              reflectivity = static_cast<val_t>(atof(argv[++i]));
  836          } else if (strcmp(argv[i], "-v") == 0) {
  837              validate = true;
  838          } else if (strcmp(argv[i], "-o") == 0) {
  839              printResults = true;
  840          } else if (strcmp(argv[i], "-h") == 0) {
  841              printUsage(argv[0]);
  842              return 0;
  843          } else {
  844              printf("Unknown option: %%s\n", argv[i]);
  845              printUsage(argv[0]);
  846              return 1;
  847          }
  848      }
  849  
  850      int subdivisions = getSubdivisionsForTriangleCount(targetTriangles);
  851  
  852      printf("Room Response Simulation Benchmark\n");
  853      printf("===================================\n");
  854      printf("Target triangles: %%d (using %%d subdivisions)\n", targetTriangles, subdivisions);
  855      printf("Timesteps: %%d\n", timesteps);
  856      printf("Source triangle: %%d\n", sourceIdx);
  857      printf("Reflectivity: %%.2f\n", reflectivity);
  858      printf("Validation: %%s\n", validate ? "enabled" : "disabled");
  859      printf("\n");
  860  
  861      // Initialize
  862      SimulationState state;
  863      initializeSimulation(state, subdivisions, static_cast<size_t>(timesteps),
  864                           static_cast<size_t>(sourceIdx), reflectivity);
  865  
  866      printf("\n");
  867  
  868      // Precomputation
  869      auto startPre = std::chrono::high_resolution_clock::now();
  870  
  871      computeTimeDelays(state);
  872      computeFormFactors(state);
  873  
  874      auto endPre = std::chrono::high_resolution_clock::now();
  875      auto preDuration = std::chrono::duration_cast<std::chrono::milliseconds>(endPre - startPre).count();
  876  
  877      printf("Precomputation time: %%ld ms\n", preDuration);
  878      printf("\n");
  879  
  880      // Simulation
  881      auto startSim = std::chrono::high_resolution_clock::now();
  882  
  883      runSimulation(state);
  884  
  885      auto endSim = std::chrono::high_resolution_clock::now();
  886      auto simDuration = std::chrono::duration_cast<std::chrono::milliseconds>(endSim - startSim).count();
  887  
  888      printf("Simulation time: %%ld ms\n", simDuration);
  889      printf("\n");
  890  
  891      // Distance computation
  892      auto startDist = std::chrono::high_resolution_clock::now();
  893  
  894      computeDistances(state);
  895  
  896      auto endDist = std::chrono::high_resolution_clock::now();
  897      auto distDuration = std::chrono::duration_cast<std::chrono::milliseconds>(endDist - startDist).count();
  898  
  899      printf("Distance computation time: %%ld ms\n", distDuration);
  900      printf("\n");
  901  
  902      // Total time
  903      long totalTime = preDuration + simDuration + distDuration;
  904      printf("Total computation time: %%ld ms\n", totalTime);
  905  
  906      // Performance metrics
  907      size_t n = state.numTriangles;
  908      size_t t = state.numTimesteps;
  909      double kijOps = static_cast<double>(n * n);
  910      double simOps = static_cast<double>(n * n * t);
  911      double distOps = static_cast<double>(n * t * t);
  912  
  913      printf("\nPerformance:\n");
  914      printf("  Triangles: %%zu\n", n);
  915      printf("  Timesteps: %%zu\n", t);
  916      printf("  Form factor computations: %%.2e\n", kijOps);
  917      printf("  Simulation operations: %%.2e\n", simOps);
  918      printf("  Distance computations: %%.2e\n", distOps);
  919      printf("  Total time per triangle: %%.4f ms\n", static_cast<double>(totalTime) / n);
  920  
  921      // Memory usage
  922      size_t memKij = n * n * sizeof(val_t);
  923      size_t memTau = n * n * sizeof(int);
  924      size_t memRad = 2 * t * n * sizeof(val_t);
  925      size_t totalMem = memKij + memTau + memRad;
  926      printf("  Memory usage: %%.2f MB\n", totalMem / (1024.0 * 1024.0));
  927  
  928      // Hash
  929      uint64_t hash = computeHash(state);
  930      printf("  Result hash: %%016lX\n", hash);
  931      printf("\n");
  932      
  933      // Print results for external validation
  934      if (printResults) {
  935          // Convert distances to double for output
  936          std::vector<double> distData(state.distances.begin(), state.distances.end());
  937          print_results(distData, "Distances");
  938      }
  939  
  940      // Validation
  941      if (validate) {
  942          if (!validateResults(state)) {
  943              return 1;
  944          }
  945      }
  946  
  947      return 0;
  948  }
